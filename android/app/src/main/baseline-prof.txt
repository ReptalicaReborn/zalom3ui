# Baseline profile: classes and methods ART should compile ahead of time, so a cold start does not run them
# interpreted. Sideloaded APKs get no Play Store cloud profile, and androidx.profileinstaller installs this one
# (merged with the libraries' own profiles: Compose, Material 3, navigation, lifecycle, activity) on first launch.
# ART compiles it at the next background dexopt (the phone or watch idle and charging).
#
# Format: https://developer.android.com/topic/performance/baselineprofiles/manually-create-measure
# H = hot, S = startup, P = post-startup; ** matches any package depth or any method/signature.
# R8 rewrites these rules to the minified names.

# All of the app's own code: it is small, and startup touches most of it (Application, MainActivity, AppViewModel,
# the transports and JSON models, the theme, App/AppNav, the chat list and chat screens).
HSPLcom/zalom3ui/app/**;->**(**)**
Lcom/zalom3ui/app/**;

# Libraries on the startup path that ship no profile of their own.
# JSON decoding of /api/status and /api/threads (kotlinx.serialization).
HSPLkotlinx/serialization/json/**;->**(**)**
Lkotlinx/serialization/json/**;
HSPLkotlinx/serialization/internal/**;->**(**)**
Lkotlinx/serialization/internal/**;
# Coroutines (viewModelScope, Dispatchers, flows on the first refresh).
HSPLkotlinx/coroutines/**;->**(**)**
Lkotlinx/coroutines/**;
# Avatars on the chat list: Coil and the OkHttp client it shares.
HSPLcoil/**;->**(**)**
Lcoil/**;
HSPLokhttp3/**;->**(**)**
Lokhttp3/**;
HSPLokio/**;->**(**)**
Lokio/**;
